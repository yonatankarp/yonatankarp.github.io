{{- /* A diagram's text alternative comes from its fence, ```mermaid {title="…" desc="…"} (#191): the title names the region and its zoom, unique on the page, and Mermaid writes both into the SVG as accTitle and accDescr, its and <desc>. One line each, with no semicolon, which ends a statement in Mermaid. */ -}} {{- $title := .Attributes.title -}} {{- $desc := .Attributes.desc -}} {{- if not (and $title $desc) -}} {{- errorf "%s: give the Mermaid diagram a title and a description, as ```mermaid {title=\"…\" desc=\"…\"}." .Position -}} {{- end -}} {{- if findRE `[;\n]` (printf "%s%s" $title $desc) -}} {{- errorf "%s: a Mermaid diagram's title and description are one line each, with no semicolon." .Position -}} {{- end -}} {{- if in (.Page.Store.Get "diagramTitles" | default slice) $title -}} {{- errorf "%s: another Mermaid diagram on this page is titled %q." .Position $title -}} {{- end -}} {{- .Page.Store.Add "diagramTitles" (slice $title) -}} {{- /* Drawn at build time (#250): scripts/render-diagrams.js builds this same source from the fence, draws it and saves it as assets/diagrams/<its sha256>.svg, which is inlined here, so the page ships the finished svg and no Mermaid. */ -}} {{- $lines := split (strings.TrimSpace .Inner) "\n" -}} {{- $src := delimit ((slice (index $lines 0) (printf " accTitle: %s" $title) (printf " accDescr: %s" $desc)) | append (after 1 $lines)) "\n" -}} {{- with resources.Get (printf "diagrams/%s.svg" (sha256 $src)) -}} <pre class="mermaid" role="region" aria-label="{{ printf hugo.Data.ui.diagram $title }}">{{ .Content | safeHTML }}</pre> {{- else -}} {{- errorf "%s: the Mermaid diagram %q is not drawn yet, or changed since it was. Run npm run build:building (or node scripts/render-diagrams.js)." .Position $title -}} {{- end }} {{ .Page.Store.Set "hasDiagram" true }}
