{{- /* The visitor's time of day for the art (#151), as statements that set html[data-time] to dawn, day, dusk or night from the sun's elevation where their time zone is (assets/js/time-pick.js, and sun.js). ?time=dawn|day|dusk|night overrides it; anything else means the sun. Shared by the homepage's pre-paint script (layouts/index.html), which also preloads that hour's city, and every other page's (head.html), whose backdrop custom.css picks by it. Returns plain JS: the caller minifies it and allows it in the CSP by its hash. ZONES, its zone table, is data/timezones.json with the links resolved, to whole degrees and grouped by the zone's folder ("Europe/": "... Berlin53,13 ..."): about 8 KB, 4.5 KB gzipped. */ -}} {{- $tz := hugo.Data.timezones }} {{- $z := newScratch }} {{- /* The links skip a retired alias the site leaves out (as scripts/lib.js's zoneTable() does). */}} {{- range $name, $to := $tz.links }}{{ if ne $name "Asia/Tel_Aviv" }}{{ $z.SetInMap "at" $name (index $tz.zones $to) }}{{ end }}{{ end }} {{- range $name, $at := merge ($z.Get "at") $tz.zones }} {{- $p := path.Split $name }} {{- $z.Set $p.Dir (trim (printf "%s %s%d,%d" ($z.Get $p.Dir | default "") $p.File (int (math.Round (index $at 1))) (int (math.Round (index $at 2)))) " ") }} {{- end }} {{- $z.Delete "at" }} {{- /* Then sunAt, from assets/js/sun.js: a module, for web3d/src/outside.js, so its export goes. Last, so the script still opens on the zone table, which check-generated-html.js reads. */}} {{ return printf "%s%s" (replace (resources.Get "js/time-pick.js").Content "ZONES" (jsonify $z.Values) 1) (replace (resources.Get "js/sun.js").Content "export " "" 1) }}
